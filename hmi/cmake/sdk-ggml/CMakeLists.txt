# Adapter for upstream's independently configured SDK. Preserve the upstream
# sources while selecting a dynamic single-ISA CPU or the portable variant set.
if(NOT EXISTS "${HMI_GGML_UPSTREAM_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "HMI_GGML_UPSTREAM_DIR must point to the bundled ggml source")
endif()
set(_hmi_profiles portable x64 sse42 avx avx2 avx-vnni avx512 avx512-vnni)
if(NOT HMI_SDK_CPU_PROFILE IN_LIST _hmi_profiles)
  message(FATAL_ERROR "Unknown HMI_SDK_CPU_PROFILE=${HMI_SDK_CPU_PROFILE}")
endif()
set(GGML_NATIVE OFF CACHE BOOL "Keep dispatcher and score baseline compatible" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "Use compatibility-checked dynamic backends" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "Expose matching diagnostic import libraries" FORCE)
# The bundled native subset contains only the backends used by this Windows app.
foreach(_backend BLAS CANN HEXAGON HIP METAL MUSA OPENCL OPENVINO RPC SYCL VIRTGPU WEBGPU ZDNN ZENDNN CPU_KLEIDIAI)
  set(GGML_${_backend} OFF)
endforeach()
foreach(_feature SSE42 AVX AVX2 BMI2 AVX_VNNI FMA F16C AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16 AMX_TILE AMX_INT8 AMX_BF16)
  set(GGML_${_feature} OFF CACHE BOOL "Explicit CPU profile feature" FORCE)
endforeach()
if(HMI_SDK_CPU_PROFILE STREQUAL "portable")
  set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "Runtime-dispatched portable CPU set" FORCE)
else()
  set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "Single guarded CPU ISA backend" FORCE)
  if(NOT HMI_SDK_CPU_PROFILE STREQUAL "x64")
    set(GGML_SSE42 ON CACHE BOOL "CPU profile" FORCE)
  endif()
  if(HMI_SDK_CPU_PROFILE MATCHES "^avx")
    set(GGML_AVX ON CACHE BOOL "CPU profile" FORCE)
  endif()
  if(HMI_SDK_CPU_PROFILE MATCHES "^avx2|^avx-vnni|^avx512")
    foreach(_feature AVX2 FMA F16C BMI2)
      set(GGML_${_feature} ON CACHE BOOL "CPU profile" FORCE)
    endforeach()
  endif()
  if(HMI_SDK_CPU_PROFILE STREQUAL "avx-vnni")
    set(GGML_AVX_VNNI ON CACHE BOOL "CPU profile" FORCE)
  endif()
  if(HMI_SDK_CPU_PROFILE MATCHES "^avx512")
    set(GGML_AVX512 ON CACHE BOOL "CPU profile" FORCE)
  endif()
  if(HMI_SDK_CPU_PROFILE STREQUAL "avx512-vnni")
    set(GGML_AVX512_VNNI ON CACHE BOOL "CPU profile" FORCE)
  endif()
endif()
# The pinned ggml project enables generic ASM even on x64 where no ASM is
# compiled. Keep its historical MSVC discovery scoped to this adapter.
set(CMAKE_POLICY_DEFAULT_CMP0194 OLD)
# The bundled MSVC CMake 4.3 ASM module queries an unpadded policy key.
set(CMAKE_POLICY_DEFAULT_CMP194 OLD)
add_subdirectory("${HMI_GGML_UPSTREAM_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/upstream")

# The optional SM89 SageAttention2 kernels currently emit uninitialized-value
# diagnostics with this SDK. Keep the established attention path by default;
# native SM89 compilation still applies to all regular CUDA kernels.
option(HMI_AUDIOCPP_SDK_SAGE_ATTN2 "Enable the optional SDK SageAttention2 kernels" OFF)
if(TARGET ggml-cuda AND NOT HMI_AUDIOCPP_SDK_SAGE_ATTN2)
  get_target_property(_hmi_cuda_sources ggml-cuda SOURCES)
  list(FILTER _hmi_cuda_sources EXCLUDE REGEX "(^|[/\\\\])sage-attn2[.]cu$")
  set_property(TARGET ggml-cuda PROPERTY SOURCES "${_hmi_cuda_sources}")
  get_target_property(_hmi_cuda_definitions ggml-cuda COMPILE_DEFINITIONS)
  list(REMOVE_ITEM _hmi_cuda_definitions GGML_CUDA_SAGE_ATTN2_ENABLED)
  set_property(TARGET ggml-cuda PROPERTY COMPILE_DEFINITIONS "${_hmi_cuda_definitions}")
endif()
